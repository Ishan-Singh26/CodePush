#include <bits/stdc++.h>
using namespace std;
 
typedef long long ll;
 
void solve() {
    int n, m;
    cin >> n >> m;
 
    vector<int> arr(n);
    vector<int> ar(m);
 
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
 
    for (int i = 0; i < m; i++) {
        cin >> ar[i];
    }
    int c1 = 0,c2 = 0;
    for(int i = 0;i<n-1;i++){
        c1 = c1+(arr[i]-arr[i+1])+1;
    }
    c1 = c1+arr[n-1];
    //
    for(int i = 0;i<m-1;i++){
        c2 = c2+(ar[i]-ar[i+1])+1;
    }
    c2 = c2+ar[m-1];
    //
    if(c1>=c2){
        cout<<1<<'\n';
    }
    else{
        cout<<2<<'\n';
    }
 
    
}
 
int main() {
    int T;
    cin >> T;
 
    while (T--) {
        solve();
    }
 
    return 0;
}
