#include <bits/stdc++.h>
using namespace std;
 
typedef long long ll;
 
void solve() {
    int a,b,n;
    cin>>a>>b>>n;
    vector<ll>arr(n);
    for(ll i = 0;i<n;i++){
        cin>>arr[i];
    }
    ll sm = 0;
    ll cnt = 0;
    for(ll i = 0;i<n;i++){
        if(arr[i]>=a){
            sm = sm+a;
            cnt++;
        }
        else{
            sm = sm+arr[i];
        }
    }
    ll ans = 0;
    ans = b+sm-cnt;
    cout<<ans<<'\n';
 
}
 
int main() {
    int T;
    cin >> T;
    while (T--) {
        solve();
    }
    return 0;
}
